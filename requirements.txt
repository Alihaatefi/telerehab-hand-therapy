# Runtime dependencies. Requires Python 3.10-3.12.
#
# MediaPipe removed the `mp.solutions` API this project uses in releases after
# 0.10.14, and 0.10.14 is also the last release published with CPython 3.12
# wheels - hence the exact pin rather than a range. All MediaPipe access is
# confined to src/telerehab/vision/hand_tracker.py, so migrating to the newer
# Tasks API is a single-file change.
mediapipe==0.10.14

# Capped below 6.12: PySide6 6.12.0 corrupts the heap ("free(): invalid size") after a
# few thousand signal emissions on Python 3.10 on Linux, which aborted the test run in
# CI. 6.11.x passes the whole suite there. Raise the cap once a fixed release has been
# verified.
PySide6>=6.7,<6.12
opencv-python>=4.8
numpy>=1.24
requests>=2.31
